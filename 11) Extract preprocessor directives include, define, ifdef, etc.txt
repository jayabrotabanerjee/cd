11. Write a C program that extracts all preprocessor directives (like #include, #define,
#ifdef) from a given C source file and lists them.
Input:
#include <stdio.h>
#define PI 3.14
int main() { return 0; }
Output:
#include <stdio.h>
#define PI 3.14

/* extract_preproc.c */
#include <stdio.h>
#include <ctype.h>

int main(int argc,char **argv){
    if(argc<2){ printf("Usage: %s file.c\n", argv[0]); return 1; }
    FILE *f=fopen(argv[1],"r"); if(!f){ perror("fopen"); return 1; }
    char line[1024];
    while(fgets(line, sizeof(line), f)){
        int i=0; while(isspace(line[i])) i++;
        if(line[i]=='#') printf("%s", line);
    }
    fclose(f);
    return 0;
}
