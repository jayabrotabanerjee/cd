14. Write a C program to analyze the frequency of each character (letters, digits, symbols)
in a source code file.
This helps understand how often certain symbols (like {, }, ;, (, )) appear in source
text.
'{' : 4
'}' : 4
'(' : 5
')' : 5
';' : 10

/* char_freq.c */
#include <stdio.h>

int main(int argc,char **argv){
    if(argc<2){ printf("Usage: %s file.c\n", argv[0]); return 1; }
    FILE *f=fopen(argv[1],"r"); if(!f){ perror("fopen"); return 1; }
    long freq[256]={0};
    int c;
    while((c=fgetc(f))!=EOF) freq[(unsigned char)c]++;
    for(int i=0;i<256;i++){
        if(freq[i]>0){
            if(i==' ') printf("' ' : %ld\n", freq[i]);
            else if(i=='\n') printf("\\n : %ld\n", freq[i]);
            else printf("'%c' : %ld\n", i, freq[i]);
        }
    }
    fclose(f);
    return 0;
}
