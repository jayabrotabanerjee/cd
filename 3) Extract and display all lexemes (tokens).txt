3. Write a C program to extract and display all lexemes (tokens) from a given C source
code.
For example,
Input: int a = b + 10;
Output: int, a, =, b, +, 10, ;

/* extract_tokens.c */
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main(int argc,char **argv){
    if(argc<2){ printf("Usage: %s file.c\n", argv[0]); return 1; }
    FILE *f=fopen(argv[1],"r"); if(!f){ perror("fopen"); return 1; }
    int c;
    char buf[256]; int i;
    int first=1;
    while((c=fgetc(f))!=EOF){
        if(isalpha(c) || c=='_'){
            i=0; buf[i++]=c;
            while((c=fgetc(f))!=EOF && (isalnum(c)||c=='_')) buf[i++]=c;
            buf[i]=0; if(!first) printf(", "); printf("%s", buf); first=0; if(c==EOF) break; ungetc(c,f);
        } else if(isdigit(c)){
            i=0; buf[i++]=c;
            while((c=fgetc(f))!=EOF && (isdigit(c)||c=='.')) buf[i++]=c;
            buf[i]=0; if(!first) printf(", "); printf("%s", buf); first=0; if(c==EOF) break; ungetc(c,f);
        } else if(isspace(c)) continue;
        else {
            char two[3]={0};
            int nxt=fgetc(f);
            two[0]=c; two[1]=nxt; two[2]=0;
            /* check two-char operators */
            if(nxt!=EOF && (strstr("== != <= >= && || ++ -- += -= *= /=", two) != NULL)){
                if(!first) printf(", "); printf("%s", two); first=0;
            } else {
                if(nxt!=EOF) ungetc(nxt,f);
                if(!first) printf(", "); printf("%c", c); first=0;
            }
        }
    }
    printf("\n");
    fclose(f);
    return 0;
}
