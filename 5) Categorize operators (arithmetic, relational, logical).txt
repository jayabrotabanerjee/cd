5. Write a C program that identifies and categorizes operators into:
 Arithmetic (+, -, *, /, %)
 Relational (==, !=, <, >, <=, >=)
 Logical (&&, ||, !)

/* categorize_ops.c */
#include <stdio.h>
#include <string.h>

int main(int argc,char **argv){
    if(argc<2){ printf("Usage: %s file.c\n", argv[0]); return 1; }
    FILE *f=fopen(argv[1],"r"); if(!f){ perror("fopen"); return 1; }
    char op[3]; int c;
    while((c=fgetc(f))!=EOF){
        if(strchr("+-*/%=&|!<>", c)){
            op[0]=c; op[1]=fgetc(f); op[2]=0;
            /* try two-char */
            if(op[1]==EOF) op[1]=0;
            int printed=0;
            /* check two-char relational/logical */
            if((op[0]=='=' && op[1]=='=') || (op[0]=='!' && op[1]=='=') || (op[0]=='<' && op[1]=='=') || (op[0]=='>' && op[1]=='=') ){
                printf("%s : Relational\n", op); printed=1;
            } else if((op[0]=='&' && op[1]=='&') || (op[0]=='|' && op[1]=='|')){
                printf("%s : Logical\n", op); printed=1;
            } else {
                if(op[1]) ungetc(op[1], f);
                char single[2]={op[0],0};
                if(strchr("+-*/%", op[0])) printf("%s : Arithmetic\n", single);
                else if(op[0]=='!' ) printf("%s : Logical\n", single);
                else if(op[0]=='=' || op[0]=='<' || op[0]=='>') printf("%s : Relational\n", single);
                else printf("%s : Other\n", single);
            }
        }
    }
    fclose(f);
    return 0;
}
