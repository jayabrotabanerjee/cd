6. Write a C program that recognizes integer, floating-point, and character constants in a
given line of code.
Example:
Input: int x = 45; float y = 4.5; char c = 'a';
Output:
Integer: 45, Float: 4.5, Character: 'a'

/* const_recognize.c */
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_integer(const char *s){
    int i=0; if(s[0]=='+'||s[0]=='-') i++;
    for(; s[i]; ++i) if(!isdigit(s[i])) return 0;
    return i>0;
}
int is_float(const char *s){
    int i=0, dot=0; if(s[0]=='+'||s[0]=='-') i++;
    for(; s[i]; ++i){
        if(s[i]=='.'){ dot++; if(dot>1) return 0; continue; }
        if(!isdigit(s[i])) return 0;
    }
    return dot==1;
}
int main(){
    char tok[256];
    while(scanf("%255s", tok)==1){
        if(tok[0]=='\'' && tok[strlen(tok)-1]=='\'' ) printf("Character: %s\n", tok);
        else if(is_integer(tok)) printf("Integer: %s\n", tok);
        else if(is_float(tok)) printf("Float: %s\n", tok);
        else printf("Other: %s\n", tok);
    }
    return 0;
}
