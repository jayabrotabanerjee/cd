13. Write a C program to extract and print all multi-line comments (/* ... */) from a given
C source file.
If no multi-line comments are present, print “No multi-line comments found.”

/* extract_multiline_comments.c */
#include <stdio.h>
#include <string.h>

int main(int argc,char **argv){
    if(argc<2){ printf("Usage: %s file.c\n", argv[0]); return 1; }
    FILE *f=fopen(argv[1],"r"); if(!f){ perror("fopen"); return 1; }
    int c, d;
    int found=0;
    while((c=fgetc(f))!=EOF){
        if(c=='/' && (d=fgetc(f))=='*'){
            found=1;
            putchar('/');
            putchar('*');
            while((c=fgetc(f))!=EOF){
                putchar(c);
                if(c=='*' && (d=fgetc(f))=='/'){ putchar(d); break; }
                if(c==EOF) break;
                if(c=='\\'){ int e=fgetc(f); if(e!=EOF) putchar(e); }
            }
            putchar('\n');
        } else if(d!=EOF) ungetc(d,f);
    }
    if(!found) printf("No multi-line comments found.\n");
    fclose(f);
    return 0;
}
